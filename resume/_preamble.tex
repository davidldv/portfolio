% T1 makes accented chars single glyphs. Under the OT1 default, "Londoño"
% renders as an accent stacked on "n" and ATS parsers extract "London~o".
\usepackage[T1]{fontenc}
% Latin Modern is the scalable Type1 build of Computer Modern. Without it the
% document ships bitmap Type3 fonts, which some ATS parsers cannot read at all.
\usepackage{lmodern}
% No f-ligatures: at 10.5pt the visual difference is invisible, and it
% guarantees "verification"/"specific" survive an ATS keyword search.
% expansion/protrusion off; we want microtype only for the ligature control.
\usepackage[expansion=false, protrusion=false]{microtype}
\DisableLigatures[f]{encoding = T1}
\usepackage{tabularx}
\usepackage{multicol}

\input{glyphtounicode}
% ATS parsers read the PDF text layer, so every glyph needs a Unicode mapping.
% glyphtounicode.tex misses these, which is why "Londoño" used to extract as
% "London?o" and "verification" came out as "verication".
\pdfglyphtounicode{ntilde}{00F1}
\pdfglyphtounicode{oacute}{00F3}
\pdfglyphtounicode{aacute}{00E1}
\pdfglyphtounicode{eacute}{00E9}
\pdfglyphtounicode{iacute}{00ED}
\pdfglyphtounicode{uacute}{00FA}
% f-ligatures extract as nothing at all, silently deleting the letters from
% keyword searches. Disabled below via microtype; mapped here as a belt-and-braces.
\pdfglyphtounicode{fi}{0066 0069}
\pdfglyphtounicode{fl}{0066 006C}
\pdfglyphtounicode{ff}{0066 0066}
\pdfglyphtounicode{ffi}{0066 0066 0069}
\pdfglyphtounicode{ffl}{0066 0066 006C}
